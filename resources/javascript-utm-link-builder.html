export function buildUtmUrl(inputUrl, campaign) {
  const url = new URL(inputUrl);
  if (url.protocol !== "https:" && url.protocol !== "http:") {
    throw new TypeError("Only HTTP and HTTPS URLs are supported.");
  }

  const fields = ["source", "medium", "campaign", "term", "content"];
  for (const field of fields) {
    const value = campaign[field];
    if (value !== undefined && String(value).trim()) {
      url.searchParams.set(`utm_${field}`, String(value).trim());
    }
  }

  return url.toString();
}

Installation guide

  1. Download the JavaScript file.
  2. Import buildUtmUrl into a browser module or copy the function.
  3. Test with a non-sensitive URL and your team's UTM naming convention.

Examples

  • buildUtmUrl('https://example.com', {source:'newsletter', medium:'email', campaign:'launch'})
